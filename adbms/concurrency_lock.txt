Part 1: Create Table and Insert Data
CREATE TABLE accounts (
    id INT PRIMARY KEY,
    name VARCHAR(50),
    balance INT
);

INSERT INTO accounts VALUES
(1, 'Priya', 5000),
(2, 'Amit', 3000);

Part 2: Check Initial Data
SELECT * FROM accounts;

Part 3: Simulating Row-Level Lock (Exclusive Lock)
Session 1
BEGIN;

UPDATE accounts
SET balance = balance - 1000
WHERE id = 1;
Do not commit yet.

Session 2
BEGIN;

UPDATE accounts
SET balance = balance + 500
WHERE id = 1;
This query will wait because Session 1 holds the lock.

Session 1 (Release Lock)
COMMIT;

Part 4: Shared Lock using SELECT FOR UPDATE
Session 1
BEGIN;

SELECT * FROM accounts
WHERE id = 2
FOR UPDATE;

Session 2
UPDATE accounts
SET balance = balance + 100
WHERE id = 2;
This query will wait.

Session 1 (Release Lock)
COMMIT;

Part 5: Demonstrating Deadlock
Session 1
BEGIN;

UPDATE accounts
SET balance = balance - 500
WHERE id = 1;
Session 2
BEGIN;

UPDATE accounts
SET balance = balance - 200
WHERE id = 2;
Session 1
UPDATE accounts
SET balance = balance - 100
WHERE id = 2;
Session 2
UPDATE accounts
SET balance = balance - 100
WHERE id = 1;
PostgreSQL will display:
ERROR: deadlock detected
