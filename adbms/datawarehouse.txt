-- Step 1: Customer Dimension
CREATE TABLE dim_customer (
 customer_id INT PRIMARY KEY,
 customer_name VARCHAR(50),
 city VARCHAR(50)
);
-- Step 2: Product Dimension
CREATE TABLE dim_product (
 product_id INT PRIMARY KEY,
 product_name VARCHAR(50),
 category VARCHAR(50)
);
-- Step 3: Time Dimension
CREATE TABLE dim_time (
 time_id INT PRIMARY KEY,
 day INT,
 month INT,
 year INT
);
-- Step 4: Sales Fact Table
CREATE TABLE fact_sales (
 sales_id SERIAL PRIMARY KEY,
 customer_id INT REFERENCES dim_customer(customer_id),
 product_id INT REFERENCES dim_product(product_id),
 time_id INT REFERENCES dim_time(time_id),
 amount DECIMAL(10,2)
);

-- Step 7: Total Sales by Customer
SELECT t.month, SUM(f.amount) AS total_sales
FROM fact_sales f
JOIN dim_time t ON f.time_id = t.time_id
GROUP BY t.month;
-- Step 8: Sales by Product Category
SELECT p.category, SUM(f.amount) AS total_sales
FROM fact_sales f
JOIN dim_product p ON f.product_id = p.product_id
GROUP BY p.category;
-- Step 9: Sales By Time (Month-Wise)
SELECT c.customer_name, SUM(f.amount) AS total_sales
FROM fact_sales f
JOIN dim_customer c ON f.customer_id = c.customer_id
GROUP BY c.customer_name;
