Step 1:
-- Run the first CREATE DATABASE on the server designated as Site 1
CREATE DATABASE site1; 
-- Run the second CREATE DATABASE on the server designated as Site 2
CREATE DATABASE site2; 

Step 2:

-- Run the `CREATE TABLE employee` and its `INSERT` statements on Site 1.

CREATE TABLE employee (
    emp_id INT PRIMARY KEY,
    emp_name VARCHAR(50),
    dept_id INT,
    salary INT 
); 


INSERT INTO employee VALUES
    (1, 'Priya', 101, 50000),
    (2, 'Rahul', 102, 60000),
    (3, 'Sneha', 101, 55000); 


-- Run the `CREATE TABLE department` and its `INSERT` statements on Site 2.


CREATE TABLE department (
    dept_id INT PRIMARY KEY,
    dept_name VARCHAR(50),
    location VARCHAR(50)
); 

INSERT INTO department VALUES
    (101, 'IT', 'Mumbai'),
    (102, 'HR', 'Pune'); 


-- Run Steps 3-7 on Site 1 (the site that will query the remote Site 2 data).
Step 3:
CREATE EXTENSION postgres_fdw; 


Step 4:
CREATE SERVER site2_server
FOREIGN DATA WRAPPER postgres_fdw
OPTIONS (host 'localhost', dbname 'site2', port '5432'); 

Step 5:
CREATE USER MAPPING FOR CURRENT_USER
SERVER site2_server
OPTIONS (user 'postgres', password 'your_password'); 

Step 6:
CREATE FOREIGN TABLE department_fdw (
    dept_id INT,
    dept_name VARCHAR(50),
    location VARCHAR(50)
)
SERVER site2_server
OPTIONS (table_name 'department'); 

-- The SELECT query should be executed on Site 1 after the FDW is configured.
Step 7:
SELECT e.emp_name, e.salary, d.dept_name, d.location
FROM employee e
JOIN department_fdw d
ON e.dept_id = d.dept_id;
