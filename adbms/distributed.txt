Step 3:
CREATE EXTENSION postgres_fdw; 

Step 4:
CREATE SERVER site2_server
FOREIGN DATA WRAPPER postgres_fdw
OPTIONS (host 'localhost', dbname 'site2', port '5432'); 

Step 5:
CREATE USER MAPPING FOR CURRENT_USER
SERVER site2_server
OPTIONS (user 'postgres', password 'your_password'); 

Step 6:
CREATE FOREIGN TABLE department_fdw (
    dept_id INT,
    dept_name VARCHAR(50),
    location VARCHAR(50)
)
SERVER site2_server
OPTIONS (table_name 'department'); 

Step 7:
SELECT e.emp_name, e.salary, d.dept_name, d.location
FROM employee e
JOIN department_fdw d
ON e.dept_id = d.dept_id;
